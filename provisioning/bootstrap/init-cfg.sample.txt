# Bootstrap init-cfg.txt for VM-Series → Strata Cloud Manager onboarding.
# Rendered per device (substitute ${...}) and delivered via the bootstrap package
# (AWS: user-data or an S3 bootstrap bucket). The rendered file carries the
# registration PIN — NEVER commit rendered output (.gitignore excludes
# *.rendered.*). This template is safe; the values are not.
#
# SCM onboarding keys CONFIRMED from Palo docs (device sub-spike, 2026-07-23):
#   docs.paloaltonetworks.com/vm-series → Create the init-cfg.txt File
#   docs.paloaltonetworks.com/strata-cloud-manager → Onboard to Strata Cloud Manager

type=dhcp-client
hostname=${HOSTNAME}
dns-primary=${DNS_PRIMARY}
dns-secondary=${DNS_SECONDARY}
dhcp-accept-server-hostname=yes

# ── Strata Cloud Manager onboarding ──────────────────────────────────────
# panorama-server=cloud is the literal that points the firewall at SCM (initiates
# TLS to the cloud management service edge — NOT a Panorama IP).
panorama-server=cloud

# The auto-registration PIN (generated in SCM → device onboarding / setup) requests
# a Thermite device certificate that authenticates this firewall to your tenant.
# PREREQUISITE: generate the PIN in SCM before boot; it is time-limited.
vm-series-auto-registration-pin-id=${SCM_REGISTRATION_PIN_ID}
vm-series-auto-registration-pin-value=${SCM_REGISTRATION_PIN_VALUE}

# dgname = the target SCM FOLDER. SCM prioritises this over any onboarding rule,
# so the device lands in our folder deterministically.
dgname=${SCM_FOLDER}   # e.g. GitOps

# Optional cloud/VM-Series toggles:
#   op-command-modes=jumbo-frame
#   plugin-op-commands=advance-routing:enable

# ── BYOL licensing ───────────────────────────────────────────────────────
# For BYOL bootstrap, place the auth code in the bootstrap package under
# /license/authcodes (a separate file, NOT this init-cfg). The firewall
# activates the license from it on first boot.

#   profile) from your Palo licensing / CSP.
